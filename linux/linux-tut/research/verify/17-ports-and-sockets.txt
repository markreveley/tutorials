### 8. wait-for-port.sh (shellcheck clean)
#!/usr/bin/env bash
# wait-for-port.sh HOST PORT [DEADLINE_SECONDS] -- exit 0 when HOST:PORT accepts a TCP connection
set -u
host=$1 port=$2 deadline=${3:-30}
start=$SECONDS
while (( SECONDS - start < deadline )); do
  if timeout 2 bash -c "echo > /dev/tcp/$host/$port" 2>/dev/null; then
    echo "open after $(( SECONDS - start ))s"
    exit 0
  fi
  sleep 1
done
echo "still not accepting connections on $host:$port after ${deadline}s" >&2
exit 1
