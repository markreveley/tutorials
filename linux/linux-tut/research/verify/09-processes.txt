# Three background jobs in a terminal; then the terminal goes away.
tmux new-session -d -s demo
tmux send-keys -t demo 'sleep 800 &' Enter
tmux send-keys -t demo 'nohup sleep 900 >/dev/null 2>&1 &' Enter

--- toolcall.py:
import subprocess, time

def tool_call(cmd, timeout=5):
    """What a harness does: run one command, capture both streams."""
    t0 = time.time()
    try:
        r = subprocess.run(["bash", "-c", cmd], capture_output=True,
                           text=True, timeout=timeout)
        out = f"exit={r.returncode} stdout={r.stdout!r}"
    except subprocess.TimeoutExpired:
        out = "TIMEOUT"
    print(f"{time.time() - t0:4.1f}s  {out}  <- {cmd}")

tool_call("sleep 3 & echo started")
tool_call("sleep 3 >/dev/null 2>&1 & echo started")
tool_call("sleep 30 & echo started")
