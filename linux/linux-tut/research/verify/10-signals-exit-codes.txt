wait $!

--- account.py:
import json, sys

rows = [json.loads(line) for line in open(sys.argv[1])]
RUN_OUTCOMES = {124: "timeout", 137: "killed"}   # about the run, not the answer

scheduled = len(rows)
not_completed = [r for r in rows if r["exit_code"] in RUN_OUTCOMES]
completed = scheduled - len(not_completed)
correct = sum(r["passed"] for r in rows)

print(f"scheduled {scheduled}  completed {completed}  correct {correct}")
for code, name in RUN_OUTCOMES.items():
    print(f"  {name} ({code}): {sum(r['exit_code'] == code for r in rows)}")
print(f"correct/scheduled   = {correct / scheduled:.3f}")
print(f"correct/completed   = {correct / completed:.3f}")
print(f"completed/scheduled = {completed / scheduled:.3f}")
