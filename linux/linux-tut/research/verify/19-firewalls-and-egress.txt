#!/usr/bin/env bash
# probe.sh [HOST_PORT] [PEER_IP:PORT] -- from inside a sandbox: what can this box reach?
set -u
host_port=${1:-8000}; peer=${2:-}
echo "== interfaces (fallback tunnel devices hidden) and routes"
ip -br addr | grep -v '@NONE'
ip route
# gateway = default route's next hop; with no default route, assume .1 of the first subnet
gw=$(ip route | awk '/^default/ {print $3; exit}')
[ -n "$gw" ] || gw=$(ip route | awk '/proto kernel/ {split($1, a, "[./]"); print a[1] "." a[2] "." a[3] ".1"; exit}')
echo "== targets"
try() {  # try LABEL URL [curl options...]: any HTTP answer counts as reachable
  local label=$1 url=$2 code rc; shift 2
  code=$(curl -sS -o /dev/null -m 5 --connect-timeout 3 -w '%{http_code}' "$@" "$url" 2>/dev/null)
  rc=$?
  if [ "$rc" -eq 0 ]; then echo "REACHABLE  $label  (HTTP $code)"
  else echo "blocked    $label  (curl exit $rc)"; fi
}
if getent ahosts example.com >/dev/null; then echo "REACHABLE  DNS: external names resolve"
else echo "blocked    DNS: external names do not resolve"; fi
try "internet https://example.com" https://example.com/
try "internet by IP, IPv4 1.1.1.1" http://1.1.1.1/
try "internet over IPv6" https://example.com/ -6
if [ -n "$gw" ]; then try "host service $gw:$host_port" "http://$gw:$host_port/"
else echo "blocked    host service (no gateway: no interface but lo)"; fi
try "metadata 169.254.169.254" http://169.254.169.254/
if [ -n "$peer" ]; then try "neighbour $peer" "http://$peer/"; fi
