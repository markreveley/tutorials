## check-output.sh (shellcheck clean)
#!/bin/bash
# check-output.sh OUTPUT EXPECTED -- compare, but only if OUTPUT is a real regular file
out=$1 expected=$2
if [ -L "$out" ]; then
  echo "FAIL: $out is a symlink to $(readlink -f "$out")"; exit 1
fi
if [ ! -f "$out" ]; then
  echo "FAIL: $out is not a regular file"; exit 1
fi
cmp -s "$out" "$expected" && echo PASS || { echo "FAIL: wrong content"; exit 1; }
